    <script>
        (function() {

            const scanBtn = document.getElementById('scanBtn');
            const urlInput = document.getElementById('urlInput');
            const resultsEl = document.getElementById('results');
            const issueList = document.getElementById('issueList');
            const scoreNumber = document.getElementById('scoreNumber');
            const criticalCount = document.getElementById('criticalCount');
            const highCount = document.getElementById('highCount');
            const mediumCount = document.getElementById('mediumCount');
            const lowCount = document.getElementById('lowCount');
            const passedCount = document.getElementById('passedCount');
            const scanTime = document.getElementById('scanTime');
            const exportBtn = document.getElementById('exportBtn');

            const API_URL = window.location.hostname === 'localhost'
                ? 'http://localhost:5000/scan'
                : '/api/scan';

            function runScan(url) {
                scanBtn.disabled = true;
                scanBtn.textContent = 'Scanning...';
                resultsEl.classList.remove('active');
                issueList.innerHTML = '';

                fetch(API_URL, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ url })
                })
                .then(res => {
                    if (!res.ok) throw new Error('Scan failed');
                    return res.json();
                })
                .then(data => {
                    renderResults(data);
                    resultsEl.classList.add('active');
                })
                .catch(err => {
                    issueList.innerHTML = `
                        <div style="color:#f55a6e;padding:16px;background:#12121a;border:1px solid #1e1e26;border-radius:8px;">
                            <strong>Error:</strong> ${err.message}
                        </div>
                    `;
                    resultsEl.classList.add('active');
                })
                .finally(() => {
                    scanBtn.disabled = false;
                    scanBtn.textContent = 'Run Scan';
                });
            }

            function renderResults(data) {
                const { results, score, summary } = data;

                const s = Math.round(score || 0);
                scoreNumber.textContent = s;
                document.querySelector('.score-ring').style.background =
                    `conic-gradient(#5f7af5 ${s}%, #1e1e26 ${s}%)`;

                criticalCount.textContent = summary?.critical || 0;
                highCount.textContent = summary?.high || 0;
                mediumCount.textContent = summary?.medium || 0;
                lowCount.textContent = summary?.low || 0;
                passedCount.textContent = summary?.passed || 0;

                scanTime.textContent = `Scanned: ${new Date().toLocaleString()}`;

                if (!results || results.length === 0) {
                    issueList.innerHTML = `
                        <div style="color:#3bcc8a;padding:16px;background:#12121a;border:1px solid #1e1e26;border-radius:8px;">
                            No vulnerabilities found.
                        </div>
                    `;
                    return;
                }

                issueList.innerHTML = results.map((item, i) => {
                    const sev = item.severity.toLowerCase();
                    const label = item.severity.charAt(0).toUpperCase() + item.severity.slice(1);
                    return `
                        <div class="issue-item ${sev}">
                            <div class="head">
                                <span class="title">${i + 1}. ${item.title}</span>
                                <span class="severity sev-${sev}">${label}</span>
                            </div>
                            <div class="description">${item.description}</div>
                            ${item.fix ? `
                                <span class="fix-label">Remediation</span>
                                <div class="fix">${item.fix}</div>
                                <button class="copy-btn" onclick="copyFix(this)">Copy fix</button>
                            ` : ''}
                        </div>
                    `;
                }).join('');
            }

            window.copyFix = function(btn) {
                const block = btn.previousElementSibling;
                if (!block) return;
                navigator.clipboard.writeText(block.textContent).then(() => {
                    btn.textContent = 'Copied';
                    setTimeout(() => { btn.textContent = 'Copy fix'; }, 2000);
                });
            };

            scanBtn.addEventListener('click', () => {
                const url = urlInput.value.trim();
                if (!url) {
                    alert('Enter a URL.');
                    return;
                }
                runScan(url);
            });

            urlInput.addEventListener('keydown', (e) => {
                if (e.key === 'Enter') scanBtn.click();
            });

            exportBtn.addEventListener('click', () => {
                window.print();
            });

        })();
    </script>
